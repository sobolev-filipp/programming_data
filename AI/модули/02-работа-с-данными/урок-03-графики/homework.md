```python
import pandas as pd, matplotlib.pyplot as plt, seaborn as sns
sns.set_theme()
df = pd.read_csv("https://raw.githubusercontent.com/datasciencedojo/datasets/master/titanic.csv").drop(columns=['Cabin'])
df['Age'] = df['Age'].fillna(df['Age'].median())
```
