## Шпаргалка (из практики)

```python
# Сеть для цифр
class DigitNet(nn.Module):
    def __init__(self):
        super().__init__()
        self.fc1 = nn.Linear(784, 128); self.fc2 = nn.Linear(128, 10)
    def forward(self, x):
        x = x.view(x.size(0), -1)        # 28×28 → 784
        x = torch.relu(self.fc1(x))
        return self.fc2(x)               # БЕЗ softmax (он в loss)

loss_fn = nn.CrossEntropyLoss()          # многоклассы
```
