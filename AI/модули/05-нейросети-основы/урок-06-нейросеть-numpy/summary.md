## Класс NeuralNetwork (скелет)

```python
class NeuralNetwork:
    def __init__(self, n_input, n_hidden, lr=0.5):
        self.W1 = np.random.randn(n_input, n_hidden) * 0.5
        self.b1 = np.zeros(n_hidden)
        self.W2 = np.random.randn(n_hidden, 1) * 0.5
        self.b2 = np.zeros(1)
        self.lr = lr

    def forward(self, X):
        self.H = sigmoid(X @ self.W1 + self.b1)
        self.out = sigmoid(self.H @ self.W2 + self.b2)
        return self.out

    def train(self, X, y, epochs=500):
        y = y.reshape(-1, 1)
        for _ in range(epochs):
            out = self.forward(X)                    # forward
            d_out = (out - y) / len(y)               # backward...
            dW2 = self.H.T @ d_out
            d_H = (d_out @ self.W2.T) * self.H*(1-self.H)
            dW1 = X.T @ d_H
            self.W1 -= self.lr*dW1; self.W2 -= self.lr*dW2   # update (+ b)
            self.b1 -= self.lr*d_H.sum(0); self.b2 -= self.lr*d_out.sum(0)

    def predict(self, X):
        return (self.forward(X) > 0.5).astype(int).ravel()
```

## Использование (как sklearn!)

```python
net = NeuralNetwork(n_input=30, n_hidden=16, lr=0.5)
```
